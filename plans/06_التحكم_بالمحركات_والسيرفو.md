## الهدف

### الكلاس: `MotorController`

```python
class MotorController:
    def __init__(self, bt_manager):
        self.bt = bt_manager
        
        # حالة المحركات الحالية
        self.state = {
            "M1": 0,          # محرك DC 1 (-255 to 255)
            "M2": 0,          # محرك DC 2 (-255 to 255)
            "S1": 90,         # سيرفو 1 (0-180°)
            "S2": 90,         # سيرفو 2 (0-180°)
        }
        
        # سرعة افتراضية
        self.default_speed = 200       # 0-255
        self.turn_speed = 150          # سرعة عند الدوران
        self.servo_center = 90         # وضع المنتصف للسيرفو
        self.servo_min = 0             # أقل زاوية
        self.servo_max = 180           # أكبر زاوية
        
        # حد السرعة
        self.max_speed = 255
        self.min_speed = -255
```

### الدوال المطلوبة

#### 1. `execute_move(data: dict) -> bool`
ينفذ أمر حركة ويطلبه من الأردوينو:

```python
def execute_move(self, data: dict) -> bool:
    """
    data يمكن أن يكون:
    - {"direction": "F", "speed": 200}  → أمر اتجاه
    - {"M1": 200, "M2": 200, "S1": 90}  → أمر مباشر
    
    الخريطة:
    - F (أمام):   M1=+speed, M2=+speed, S1=90, S2=90
    - B (خلف):    M1=-speed, M2=-speed, S1=90, S2=90
    - L (يسار):   M1=+speed, M2=+speed, S1=0, S2=0   (سيرفو لليسار)
    - R (يمين):   M1=+speed, M2=+speed, S1=180, S2=180 (سيرفو لليمين)
    - S (توقف):   M1=0, M2=0, S1=90, S2=90
    """
    
    if "direction" in data:
        direction = data["direction"]
        speed = data.get("speed", self.default_speed)
        
        if direction == "F":
            self.state = {"M1": speed, "M2": speed, "S1": 90, "S2": 90}
        elif direction == "B":
            self.state = {"M1": -speed, "M2": -speed, "S1": 90, "S2": 90}
        elif direction == "L":
            self.state = {"M1": speed, "M2": speed, "S1": 0, "S2": 0}
        elif direction == "R":
            self.state = {"M1": speed, "M2": speed, "S1": 180, "S2": 180}
        elif direction == "S":
            self.state = {"M1": 0, "M2": 0, "S1": 90, "S2": 90}
        elif direction == "FL":  # أمام-يسار
            self.state = {"M1": speed, "M2": speed, "S1": 45, "S2": 45}
        elif direction == "FR":  # أمام-يمين
            self.state = {"M1": speed, "M2": speed, "S1": 135, "S2": 135}
        elif direction == "BL":  # خلف-يسار
            self.state = {"M1": -speed, "M2": -speed, "S1": 45, "S2": 45}
        elif direction == "BR":  # خلف-يمين
            self.state = {"M1": -speed, "M2": -speed, "S1": 135, "S2": 135}
        elif direction == "SPIN_L":  # دوران في المكان يسار
            self.state = {"M1": speed, "M2": -speed, "S1": 90, "S2": 90}
        elif direction == "SPIN_R":  # دوران في المكان يمين
            self.state = {"M1": -speed, "M2": speed, "S1": 90, "S2": 90}
    else:
        # أمر مباشر - تحديث جزئي
        for key in ["M1", "M2", "S1", "S2"]:
            if key in data:
                self.state[key] = data[key]
    
    # تطبيق حدود
    self._clamp_values()
    
    # إرسال للأردوينو
    return self.bt.send_json(self.state)
```

#### 2. `set_motor_speed(motor: str, speed: int) -> bool`
يضبط سرعة محرك DC محدد:
```python
def set_motor_speed(self, motor: str, speed: int) -> bool:
    """motor: 'M1' or 'M2', speed: -255 to 255"""
    self.state[motor] = max(self.min_speed, min(self.max_speed, speed))
    return self.bt.send_json(self.state)
```

#### 3. `set_servo_angle(servo: str, angle: int) -> bool`
يضبط زاوية محرك سيرفو محدد:
```python
def set_servo_angle(self, servo: str, angle: int) -> bool:
    """servo: 'S1' or 'S2', angle: 0 to 180"""
    angle = max(self.servo_min, min(self.servo_max, angle))
    self.state[servo] = angle
    return self.bt.send_json(self.state)
```

#### 7. `_clamp_values()`
يضمن أن القيم ضمن الحدود:

```python
def _clamp_values(self):
    for motor in ["M1", "M2"]:
        self.state[motor] = max(self.min_speed, min(self.max_speed, self.state[motor]))
    for servo in ["S1", "S2"]:
        self.state[servo] = max(self.servo_min, min(self.servo_max, self.state[servo]))
```

## بروتوكول الأوامر — التفاصيل

### أنواع الأوامر

#### 1. أوامر حركة بسيطة (Direction)
```
{"CMD": "F", "SPD": 200}
{"CMD": "B", "SPD": 200}
{"CMD": "L", "SPD": 200}
{"CMD": "R", "SPD": 200}
{"CMD": "S"}
```
الأردوينو يستقبل `CMD` وينفذ الحركة المناسبة مباشرة.

#### 2. أوامر تحكم دقيق (Direct)
```
{"M1": 200, "M2": 200, "S1": 90, "S2": 90}
```
الأردوينو يضبط كل محرك بالقيم المحددة.

#### 3. أوامر سيرفو فقط
```
{"S1": 45, "S2": 45}
```
يغير اتجاه السيرفو فقط دون تغيير سرعة DC.

#### 4. أوامر سرعة فقط
```
{"M1": 150, "M2": 150}
```
يغير سرعة المحرك فقط دون تغيير اتجاه السيرفو.

## تحويل القيم للأردوينو

```python
def build_arduino_command(self) -> str:
    """يبني الأمر بصيغة يفهمها الأردوينو"""
    command = {}
    
    # DC Motors: القيمة الموجبة = أمام، سالبة = خلف
    for motor in ["M1", "M2"]:
        value = self.state[motor]
        direction = 1 if value >= 0 else 0
        speed = abs(value)
        command[motor] = speed
        command[f"{motor}D"] = direction  # Direction
    
    # Servo: زاوية مباشرة (0-180)
    for servo in ["S1", "S2"]:
        command[servo] = self.state[servo]
    
    return json.dumps(command)
```

## السلامة (Safety)

- **Heartbeat**: كل 2 ثانية، لو الأردوينو ما رد → توقف تلقائي
- **Connection Lost**: عند انقطاع البلوتوث → إرسال أمر توقف فوري
- **Emergency Stop**: زر أحمر كبير دائماً مرئي في الواجهة
- **Speed Ramp**: لا يقفز من 0 إلى 255 مباشرة — تدرج تدريجي
```python
def _ramp_speed(self, motor, target, step=10, delay=0.05):
    """زيادة/تقليل السرعة تدريجياً"""
    current = self.state[motor]
    while current != target:
        if current < target:
            current = min(current + step, target)
        else:
            current = max(current - step, target)
        self.state[motor] = current
        self.bt.send_json(self.state)
        time.sleep(delay)
```
