# 08 - إدارة الحساسات والقراءات

## الهدف
نظام مرن لإدارة حساسات متعددة، كل حساس له كود حرفي (مثل T, H, G)، مع عرض القراءات في الوقت الحقيقي وتنبيهات.

## الملف المسؤول
- `modules/sensors.py`

## تفاصيل الوحدة: `sensors.py`

### الكلاس: `SensorManager`

```python
class SensorManager:
    def __init__(self):
        self.sensors = {}            # {code: SensorObject}
        self.readings = {}           # {code: value} آخر قراءة
        self.history = {}            # {code: [(timestamp, value), ...]}
        self.history_max = 100       # أقصى عدد قراءات محفوظة
        self.callbacks = []          # دوال تُنفذ عند قراءة جديدة
        self.warning_active = {}     # {code: bool} هل التحذير مفعل؟
```

### كلاس الحساس: `Sensor`

```python
class Sensor:
    def __init__(self, code, name, unit, icon, min_val, max_val, 
                 warn_high=None, warn_low=None, color="#00b894"):
        self.code = code              # حرف واحد: T, H, G, P, D, L...
        self.name = name              # اسم الحساس: Temperature
        self.unit = unit              # وحدة القياس: °C, %, ppm, hPa
        self.icon = icon              # رمز تعبيري: 🌡️
        self.min_val = min_val        # الحد الأدنى
        self.max_val = max_val        # الحد الأقصى
        self.warn_high = warn_high    # حد التحذير العلوي
        self.warn_low = warn_low      # حد التحذير السفلي
        self.color = color            # لون العرض
        self.current_value = None     # القيمة الحالية
        self.last_update = None       # وقت آخر تحديث
```

### الدوال المطلوبة

#### 1. `register_sensor(sensor: Sensor) -> bool`
يسجل حساس جديد:
```python
def register_sensor(self, sensor: Sensor) -> bool:
    if sensor.code in self.sensors:
        return False  # الكود موجود مسبقاً
    self.sensors[sensor.code] = sensor
    self.readings[sensor.code] = None
    self.history[sensor.code] = []
    self.warning_active[sensor.code] = False
    return True
```

#### 2. `remove_sensor(code: str) -> bool`
يزيل حساس

#### 3. `update(readings: dict) -> None`
يحدث القراءات من بيانات الأردوينو:
```python
def update(self, readings: dict) -> None:
    """
    readings: {"T": 25.5, "G": 0.2, "H": 60}
    يتم استدعاؤها كل مرة تصل فيها بيانات جديدة من السيريال
    """
    for code, value in readings.items():
        if code in self.sensors:
            sensor = self.sensors[code]
            sensor.current_value = value
            sensor.last_update = datetime.now()
            
            # حفظ في التاريخ
            self.readings[code] = value
            self.history[code].append((datetime.now(), value))
            
            # اقتطاع التاريخ
            if len(self.history[code]) > self.history_max:
                self.history[code] = self.history[code][-self.history_max:]
            
            # فحص التحذيرات
            self._check_warning(code, value)
    
    # تنفيذ الكولباك
    for callback in self.callbacks:
        callback(self.readings)
```

def get_all(self) -> dict:
    return {
        code: {
            "name": sensor.name,
            "icon": sensor.icon,
            "value": sensor.current_value,
            "unit": sensor.unit,
            "last_update": str(sensor.last_update),
            "status": self._get_status(code)

#### 6. `get_chart_data(code: str) -> dict`
بيانات مناسبة لرسم بياني:
```python
def get_chart_data(self, code: str) -> dict:
    history = self.history.get(code, [])
    return {
        "labels": [t.strftime("%H:%M:%S") for t, v in history],
        "values": [v for t, v in history],
        "min": self.sensors[code].min_val,
        "max": self.sensors[code].max_val,
        "warn_high": self.sensors[code].warn_high,
        "warn_low": self.sensors[code].warn_low
    }
```

#### 7. `_check_warning(code: str, value: float)`
يفحص هل القيمة خارج الحدود:
```python
def _check_warning(self, code: str, value: float):
    sensor = self.sensors[code]
    was_warning = self.warning_active[code]
    
    is_warning = False
    if sensor.warn_high and value >= sensor.warn_high:
        is_warning = True
    if sensor.warn_low and value <= sensor.warn_low:
        is_warning = True
    
    self.warning_active[code] = is_warning
    
    # إرسال تنبيه فقط لو تغيرت الحالة
    if is_warning and not was_warning:
        socketio.emit('sensor_warning', {
            "code": code,
            "name": sensor.name,
            "value": value,
            "unit": sensor.unit,
            "type": "high" if value >= sensor.warn_high else "low"
        })
```

#### 8. `_get_status(code: str) -> str`
```python
def _get_status(self, code: str) -> str:
    if self.warning_active.get(code):
        return "warning"
    elif self.readings.get(code) is not None:
        return "normal"
    else:
        return "no_data"
```

### JavaScript: تحديث القراءات
```javascript
socket.on('sensor_data', function(data) {
    for (const [code, info] of Object.entries(data)) {
        const badge = document.getElementById(`sensor-${code}`);
        if (badge) {
            badge.querySelector('.sensor-value').textContent = 
                `${info.value}${info.unit}`;
            
            if (info.status === 'warning') {
                badge.classList.add('warning');
            } else {
                badge.classList.remove('warning');
            }
        }
    }
});

socket.on('sensor_warning', function(data) {
    showNotification(
        `⚠️ تحذير: ${data.name} = ${data.value}${data.unit}`,
        'warning'
    );
});
```

// باستخدام Chart.js (يُضاف للمشروع)
function renderSensorChart(code) {
    fetch(`/api/sensor/${code}/history`)
        .then(r => r.json())
        .then(data => {
            const ctx = document.getElementById('sensor-chart').getContext('2d');
            new Chart(ctx, {
                type: 'line',
                data: {
                    labels: data.labels,
                    datasets: [{
                        label: data.name,
                        data: data.values,
                        borderColor: data.color,
                        tension: 0.3,
                        fill: true
                    }]
                },
                options: {
                    responsive: true,
                    scales: {
                        y: {
                            min: data.min,
                            max: data.max
                        }
                    }
                }
            });
        });
}
