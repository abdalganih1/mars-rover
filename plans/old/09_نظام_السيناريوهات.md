### كلاس السيناريو: `Scenario`

```python
class Scenario:
    def __init__(self, name, description="", steps=None, loop=False):
        self.name = name
        self.description = description
        self.steps = steps or []        # [ScenarioStep, ...]
        self.loop = loop                # تكرار مستمر؟
        self.created_at = datetime.now()
        self.last_run = None
        self.run_count = 0
```

### كلاس الخطوة: `ScenarioStep`

```python
class ScenarioStep:
    def __init__(self, motors=None, duration=1000, label=""):
        self.motors = motors or {}      # {"M1": 200, "M2": 200, "S1": 90, "S2": 90}
        self.duration = duration        # بالمللي ثانية
        self.label = label              # وصف اختياري
```

#### 1. `create_scenario(name, steps, description="", loop=False) -> bool`
ينشئ سيناريو جديد:
```python
def create_scenario(self, name, steps_data, description="", loop=False):
    steps = []
    for step_data in steps_data:
        step = ScenarioStep(
            motors={
                "M1": step_data.get("M1", 0),
                "M2": step_data.get("M2", 0),
                "S1": step_data.get("S1", 90),
                "S2": step_data.get("S2", 90)
            },
            duration=step_data.get("duration", 1000),
            label=step_data.get("label", "")
        )
        steps.append(step)
    
    scenario = Scenario(name, description, steps, loop)
    self.scenarios[name] = scenario
    self.save_to_file(name)
    return True
```

#### 2. `run_scenario(name) -> bool`
يشغل سيناريو في خيط خلفية:
```python
def run_scenario(self, name) -> bool:
    if name not in self.scenarios:
        return False
    if self.is_running:
        self.stop_scenario()
    
    self.current_running = name
    self.is_running = True
    self.should_stop = False
    
    # تشغيل في خيط منفصل
    thread = threading.Thread(
        target=self._execute_scenario,
        args=(name,),
        daemon=True
    )
    thread.start()
    return True
```

#### 3. `_execute_scenario(name)`
ينفذ السيناريو خطوة بخطوة:
```python
def _execute_scenario(self, name):
    scenario = self.scenarios[name]
    scenario.last_run = datetime.now()
    scenario.run_count += 1
    
    self._log(f"بدء تنفيذ: {name} ({len(scenario.steps)} خطوة)")
    
    try:
        while not self.should_stop:
            for i, step in enumerate(scenario.steps):
                if self.should_stop:
                    break
                
                # إرسال حالة المحركات
                self.motors.execute_move(step.motors)
                
                # إرسال تقدم للواجهة
                progress = {
                    "scenario": name,
                    "step": i + 1,
                    "total": len(scenario.steps),
                    "label": step.label,
                    "motors": step.motors,
                    "duration": step.duration
                }
                socketio.emit('scenario_progress', progress)
                
                self._log(f"خطوة {i+1}/{len(scenario.steps)}: {step.label or 'حركة'}")
                
                # انتظار المدة المحددة (مع فحص الإيقاف)
                elapsed = 0
                check_interval = 50  # فحص كل 50ms
                while elapsed < step.duration and not self.should_stop:
                    time.sleep(check_interval / 1000.0)
                    elapsed += check_interval
            
            # لو مش loop، اخرج بعد أول دورة
            if not scenario.loop:
                break
    
    except Exception as e:
        self._log(f"خطأ: {str(e)}")
    
    finally:
        # توقف دائم
        self.motors.emergency_stop()
        self.is_running = False
        self.current_running = None
        socketio.emit('scenario_progress', {
            "scenario": name,
            "step": 0,
            "total": 0,
            "status": "completed"
        })
        self._log(f"انتهى: {name}")
```

```python
def save_to_file(self, name) -> bool:
    scenario = self.scenarios[name]
    filepath = os.path.join(self.scenarios_dir, f"{name}.json")
    
    data = {
        "name": scenario.name,
        "description": scenario.description,
        "loop": scenario.loop,
        "steps": [
            {
                "motors": step.motors,
                "duration": step.duration,
                "label": step.label
            }
            for step in scenario.steps
        ]
    }
    
    with open(filepath, 'w') as f:
        json.dump(data, f, indent=2, ensure_ascii=False)
    return True
```

## واجهة السيناريوهات في Frontend

### تشغيل سيناريو من القائمة
```javascript
function runScenario(name) {
    if (confirm(`تشغيل السيناريو "${name}"؟`)) {
        socket.emit('scenario_run', { name: name });
        showNotification(`🎬 جاري تشغيل: ${name}`, 'info');
    }
}

function stopScenario() {
    socket.emit('scenario_stop');
    showNotification('⏹ تم إيقاف السيناريو', 'warning');
}
```

### متابعة التقدم
```javascript
socket.on('scenario_progress', function(data) {
    if (data.status === 'completed') {
        showNotification('✅ انتهى السيناريو', 'success');
        return;
    }
    
    // تحديث شريط التقدم
    const progress = (data.step / data.total) * 100;
    document.getElementById('scenario-progress-bar').style.width = `${progress}%`;
    document.getElementById('scenario-step-text').textContent = 
        `خطوة ${data.step}/${data.total}: ${data.label}`;
});
```

### تسجيل سيناريو بالتسجيل المباشر
```javascript
let recording = false;
let recordedSteps = [];

function startRecording() {
    recording = true;
    recordedSteps = [];
    showNotification('🔴 جاري التسجيل... اضغط الأزرار الآن', 'info');
}

function stopRecording() {
    recording = false;
    showNotification('⏹ تم إيقاف التسجيل', 'info');
    
    const name = prompt('اسم السيناريو:');
    if (name) {
        socket.emit('scenario_save', {
            name: name,
            steps: recordedSteps
        });
    }
}

// تعديل دوال الأزرار لتسجيل الخطوات
function onPress(direction) {

        });
    }
}
```
