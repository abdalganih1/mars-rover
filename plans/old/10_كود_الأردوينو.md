# 10 - كود الأردوينو

## الهدف
كود Arduino يستقبل أوامر JSON من الراسبيري عبر البلوتوث HC-05 ويتحكم بالمحركات والحساسات.

## الملف
- `arduino/robot_controller.ino`

## المكتبات المطلوبة (Arduino)
```cpp
#include <SoftwareSerial.h>   // للبلوتوث HC-05
#include <Servo.h>            // لمحركات السيرفو
#include <ArduinoJson.h>      // لتحليل JSON (v6+)
```

## تعريف الأرجل (Pins)

```cpp
// === البلوتوث HC-05 ===
#define BT_RX 2     // Arduino TX → HC-05 RX
#define BT_TX 3     // Arduino RX ← HC-05 TX

// === محركات DC عبر L298N ===
#define M1_EN 5     // PWM speed control motor 1
#define M1_IN1 6    // Direction motor 1
#define M1_IN2 7    // Direction motor 1
#define M2_EN 10    // PWM speed control motor 2
#define M2_IN1 8    // Direction motor 2
#define M2_IN2 9    // Direction motor 2

// === محركات السيرفو ===
#define S1_PIN 11   // Servo 1
#define S2_PIN 12   // Servo 2

// === حساسات (أمثلة) ===
#define TEMP_PIN A0    // حساس حرارة
#define GAS_PIN A1     // حساس غاز
#define HUMIDITY_PIN A2 // حساس رطوبة
#define TRIG_PIN 4     // حساس مسافة Trigger
#define ECHO_PIN 13    // حساس مسافة Echo
```

## الكود الكامل

```cpp
#include <SoftwareSerial.h>
#include <Servo.h>
#include <ArduinoJson.h>

// === البلوتوث ===
SoftwareSerial btSerial(BT_TX, BT_RX); // RX, TX

// === السيرفو ===
Servo servo1;
Servo servo2;

// === المتغيرات العامة ===
int m1Speed = 0;
int m2Speed = 0;
int s1Angle = 90;
int s2Angle = 90;

// === إعدادات القراءة ===
unsigned long lastSensorRead = 0;
const unsigned long SENSOR_INTERVAL = 500; // قراءة كل 500ms
const int JSON_BUFFER_SIZE = 256;

void setup() {
    // Serial للـ debugging
    Serial.begin(9600);
    
    // البلوتوث
    btSerial.begin(9600);
    
    // محركات DC
    pinMode(M1_EN, OUTPUT);
    pinMode(M1_IN1, OUTPUT);
    pinMode(M1_IN2, OUTPUT);
    pinMode(M2_EN, OUTPUT);
    pinMode(M2_IN1, OUTPUT);
    pinMode(M2_IN2, OUTPUT);
    
    // السيرفو
    servo1.attach(S1_PIN);
    servo2.attach(S2_PIN);
    servo1.write(90);
    servo2.write(90);
    
    // حساس المسافة
    pinMode(TRIG_PIN, OUTPUT);
    pinMode(ECHO_PIN, INPUT);
    
    // حساسات تناظرية
    pinMode(TEMP_PIN, INPUT);
    pinMode(GAS_PIN, INPUT);
    pinMode(HUMIDITY_PIN, INPUT);
    
    Serial.println("Robot Controller Ready!");
}

void loop() {
    // 1. قراءة الأوامر من البلوتوث
    if (btSerial.available()) {
        String jsonStr = btSerial.readStringUntil('\n');
        jsonStr.trim();
        
        if (jsonStr.length() > 0) {
            processCommand(jsonStr);
        }
    }
    
    // 2. إرسال بيانات الحساسات بشكل دوري
    unsigned long now = millis();
    if (now - lastSensorRead >= SENSOR_INTERVAL) {
        sendSensorData();
        lastSensorRead = now;
    }
}

void processCommand(String jsonStr) {
    StaticJsonDocument<JSON_BUFFER_SIZE> doc;
    DeserializationError error = deserializeJson(doc, jsonStr);
    
    if (error) {
        Serial.print("JSON Error: ");
        Serial.println(error.c_str());
        return;
    }
    
    // === أمر اتجاه بسيط ===
    if (doc.containsKey("CMD")) {
        String cmd = doc["CMD"].as<String>();
        int speed = doc.containsKey("SPD") ? doc["SPD"].as<int>() : 200;
        
        if (cmd == "F") {
            setMotor(1, speed);
            setMotor(2, speed);
            servo1.write(90);
            servo2.write(90);
        }
        else if (cmd == "B") {
            setMotor(1, -speed);
            setMotor(2, -speed);
            servo1.write(90);
            servo2.write(90);
        }
        else if (cmd == "L") {
            setMotor(1, speed);
            setMotor(2, speed);
            servo1.write(0);
            servo2.write(0);
        }
        else if (cmd == "R") {
            setMotor(1, speed);
            setMotor(2, speed);
            servo1.write(180);
            servo2.write(180);
        }
        else if (cmd == "S") {
            setMotor(1, 0);
            setMotor(2, 0);
            servo1.write(90);
            servo2.write(90);
        }
        else if (cmd == "FL") {
            setMotor(1, speed);
            setMotor(2, speed);
            servo1.write(45);
            servo2.write(45);
        }
        else if (cmd == "FR") {
            setMotor(1, speed);
            setMotor(2, speed);
            servo1.write(135);
            servo2.write(135);
        }
    }
    
    // === أوامر مباشرة ===
    if (doc.containsKey("M1")) {
        setMotor(1, doc["M1"].as<int>());
    }
    if (doc.containsKey("M2")) {
        setMotor(2, doc["M2"].as<int>());
    }
    if (doc.containsKey("S1")) {
        s1Angle = constrain(doc["S1"].as<int>(), 0, 180);
        servo1.write(s1Angle);
    }
    if (doc.containsKey("S2")) {
        s2Angle = constrain(doc["S2"].as<int>(), 0, 180);
        servo2.write(s2Angle);
    }
}

void setMotor(int motor, int speed) {
    speed = constrain(speed, -255, 255);
    
    int enPin, in1Pin, in2Pin;
    
    if (motor == 1) {
        enPin = M1_EN; in1Pin = M1_IN1; in2Pin = M1_IN2;
        m1Speed = speed;
    } else {
        enPin = M2_EN; in1Pin = M2_IN1; in2Pin = M2_IN2;
        m2Speed = speed;
    }
    
    if (speed > 0) {
        // أمام
        digitalWrite(in1Pin, HIGH);
        digitalWrite(in2Pin, LOW);
        analogWrite(enPin, speed);
    }
    else if (speed < 0) {
        // خلف
        digitalWrite(in1Pin, LOW);
        digitalWrite(in2Pin, HIGH);
        analogWrite(enPin, -speed);
    }
    else {
        // توقف
        digitalWrite(in1Pin, LOW);
        digitalWrite(in2Pin, LOW);
        analogWrite(enPin, 0);
    }
}

void sendSensorData() {
    StaticJsonDocument<256> doc;
    
    // قراءة الحرارة (مثال مع LM35)
    float temp = analogRead(TEMP_PIN) * (5.0 / 1023.0) * 100.0;
    doc["T"] = round(temp * 10.0) / 10.0;
    
    // قراءة الغاز
    int gas = analogRead(GAS_PIN);
    float gasValue = gas * (5.0 / 1023.0);
    doc["G"] = round(gasValue * 100.0) / 100.0;
    
    // قراءة الرطوبة
    int hum = analogRead(HUMIDITY_PIN);
    float humidity = map(hum, 0, 1023, 0, 100);
    doc["H"] = round(humidity);
    
    // قراءة المسافة (Ultrasonic)
    float distance = readDistance();
    if (distance > 0) {
        doc["D"] = round(distance * 10.0) / 10.0;
    }
    
    // إرسال كـ JSON
    String output;
    serializeJson(doc, output);
    btSerial.println(output);
    
    // debugging
    Serial.println(output);
}

float readDistance() {
    digitalWrite(TRIG_PIN, LOW);
    delayMicroseconds(2);
    digitalWrite(TRIG_PIN, HIGH);
    delayMicroseconds(10);
    digitalWrite(TRIG_PIN, LOW);
    
    long duration = pulseIn(ECHO_PIN, HIGH, 30000); // timeout 30ms
    
    if (duration == 0) return -1;
    
    float distance = (duration * 0.0343) / 2.0;
    return distance;
}
```

## مكتبة ArduinoJson

### تثبيت
1. افتح Arduino IDE
2. Sketch → Include Library → Manage Libraries
3. ابحث عن "ArduinoJson"
4. ثبّت الإصدار 6.x أو أحدث

### ملاحظات
- `StaticJsonDocument<256>` يكفي لأغلب الأوامر
- لو عدد الحساسات كبير، زد الحجم إلى 512
- `deserializeJson` يحول النص إلى كائن JSON
- `serializeJson` يحول الكائن إلى نص

## توصيل HC-05 مع الأردوينو

```
HC-05        Arduino
─────        ───────
VCC    →     5V
GND    →     GND
TXD    →     Pin 3 (Arduino RX)
RXD    →     Pin 2 (Arduino TX) عبر مقاومة voltage divider
```

### ⚠️ مهم: Voltage Divider
الأردوينو يعمل بـ 5V لكن HC-05 RX يقبل 3.3V:
```
Arduino Pin 2 ──[1KΩ]──┬── HC-05 RXD
                        │
                    [2KΩ]
                        │
                       GND
```

## إعدادات HC-05 الافتراضية
- **Baud Rate**: 9600
- **اسم الجهاز**: HC-05
- **PIN**: 1234
- **وضع**: Slave
