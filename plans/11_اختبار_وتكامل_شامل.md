## المرحلة 2: اختبار الكاميرا

```python
# test_camera.py
import cv2

cap = cv2.VideoCapture(0)
if not cap.isOpened():
    print("❌ فشل فتح الكاميرا")
    exit()

print("✅ الكاميرا تعمل")

# التقاط إطار واحد
ret, frame = cap.read()
if ret:
    cv2.imwrite('/home/user/program/test_frame.jpg', frame)
    print(f"✅ إطار محفوظ: {frame.shape}")
    
    # اختبار التحويلات
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    cv2.imwrite('/home/user/program/test_gray.jpg', gray)
    print("✅ تحويل رمادي يعمل")
    
    _, binary = cv2.threshold(gray, 128, 255, cv2.THRESH_BINARY)
    cv2.imwrite('/home/user/program/test_binary.jpg', binary)
    print("✅ تحويل ثنائي يعمل")
else:
    print("❌ فشل التقاط إطار")

cap.release()
```

## المرحلة 3: اختبار البلوتوث

```python
# test_bluetooth.py
import bluetooth
import subprocess

# 1. فحص وجود البلوتوث
print("=== فحص البلوتوث ===")
try:
    devices = bluetooth.discover_devices(duration=5, lookup_names=True)
    print(f"✅ تم العثور على {len(devices)} جهاز:")
    for addr, name in devices:
        print(f"  - {name} ({addr})")
except Exception as e:
    print(f"❌ خطأ: {e}")

# 2. محاولة الاتصال بـ HC-05
print("\n=== اختبار الاتصال ===")
target = input("أدخل MAC Address (أو اضغط Enter للتخطي): ").strip()
if target:
    try:
        subprocess.run(['sudo', 'rfcomm', 'bind', '0', target, '1'], check=True)
        print("✅ تم ربط RFCOMM")
    except Exception as e:
        print(f"❌ خطأ: {e}")
```

## المرحلة 4: اختبار السيريال

```python
# test_serial.py
import serial
import json
import time

try:
    ser = serial.Serial('/dev/rfcomm0', baudrate=9600, timeout=2)
    print("✅ تم فتح المنفذ السيريال")
    
    # إرسال أمر
    command = {"M1": 200, "M2": 200, "S1": 90, "S2": 90}
    ser.write((json.dumps(command) + '\n').encode())
    print(f"✅ تم إرسال: {command}")
    
    # انتظار رد
    time.sleep(1)
    if ser.in_waiting:
        response = ser.readline().decode().strip()
        print(f"✅ رد: {response}")
    
    # إرسال توقف
    ser.write(b'{"CMD":"S"}\n')
    print("✅ تم إرسال أمر توقف")
    
    ser.close()
except Exception as e:
    print(f"❌ خطأ: {e}")
```

## المرحلة 7: اختبار الحساسات

```python
# test_sensors.py
import serial
import json
import time

ser = serial.Serial('/dev/rfcomm0', 9600, timeout=2)

print("جاري قراءة الحساسات لمدة 10 ثواني...")
start = time.time()

while time.time() - start < 10:
    if ser.in_waiting:
        line = ser.readline().decode().strip()
        try:
            data = json.loads(line)
            print(f"📊 {data}")
        except json.JSONDecodeError:
            print(f"⚠️ بيانات مشوهة: {line}")
    time.sleep(0.1)

ser.close()
print("✅ انتهى اختبار الحساسات")
```

## استكشاف الأخطاء الشائعة

| المشكلة | الحل |
|---------|------|
| الكاميرا سوداء | تحقق من `/dev/video0` و `gpu_mem` |
| البلوتوث لا يكتشف | `sudo hciconfig hci0 up` |
| rfcomm bind فشل | `sudo rfcomm release 0` ثم أعد المحاولة |
| السيريال مرفوض | أضف المستخدم لمجموعة `dialout` |
| بطء البث | قلل الدقة أو FPS |
| الأزرار لا تعمل على الموبايل | تأكد من `{ passive: false }` في touch events |
| التوقف لا يعمل | تأكد من `touchcancel` event |
| JSON مشوه | أضف `\n` في نهاية كل رسالة |
